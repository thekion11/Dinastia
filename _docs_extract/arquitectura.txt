Documento Técnico Maestro: Arquitectura, Depuración y Motor Táctico de Simulación
Proyecto: DEMotor: Godot 4.xÁrea: Capa Núcleo (Simulación Matemática) & Capa Visor (Renderizado 3D y Cinemática Broadcast)
1. Arquitectura de 3 Capas (Núcleo vs. Visor)
Para garantizar un juego fluido, libre de desincronizaciones y con acabado profesional, el sistema preserva la separación estricta de responsabilidades:
[ Capa 1: Núcleo Matemático ]  (nucleo/partido.gd)            │            ▼ (Emite diccionario determinista de eventos/ticks)[ Capa 2: Visor de Simulación ] (visor/MatchPlayback.gd)            │            ▼ (Interpola transformaciones 3D, IK, AnimationTree)[ Capa 3: Presentación Visual ] (ui/estadio.gd, stadium_builder.gd, Shaders, Audio)
Contrato de Coordenadas y Escala Métrica (Norma FIFA / IFAB)
• Escala global: 1.0 unidad de Godot = 1.0 metro real.• Dimensiones Oficiales: Largo (X) = 105.0m, Ancho (Z) = 68.0m.• Origen de Coordenadas (0,0,0): Punto central del círculo del medio campo.• Límites en X: [-52.5m, +52.5m].• Límites en Z: [-34.0m, +34.0m].• Altura Y: 0.0m (ras de césped) a +2.44m (larguero).
2. Correcciones Críticas del Visor (Soluciones en Godot 4)
2.1. tele_dinamica.gd
Problema resuelto: La distorsión por vista ortográfica o perspectiva gran angular (que aplastaba la geometría y rompía la anatomía de los futbolistas).Solución: Teleobjetivo (FOV ≈ 20º - 22º)
# res://ui/tele_dinamica.gdclass_name TeleDinamicaextends Camera3D@export var target_balon: Node3D@export var altura_base: float = 36.0@export var distancia_z_base: float = 46.0@export var fov_transmision: float = 21.0# Amortiguaciónvar _pos_objetivo: Vector3 = Vector3.ZEROvar _foco_actual: Vector3 = Vector3.ZEROvar _velocidad_amortiguacion: Vector3 = Vector3.ZEROfunc _ready() -> void:    projection = Camera3D.PROJECTION_PERSPECTIVE    fov = fov_transmision    near = 1.0    far = 400.0func _process(delta: float) -> void:    if not target_balon:        return        var pos_b = target_balon.global_position        # Mapeo no lineal: la cámara sigue la jugada sin salir del perímetro    var cam_x = clampf(pos_b.x * 0.72, -36.0, 36.0)    var cam_z = (pos_b.z * 0.28) + distancia_z_base    var cam_y = altura_base + (abs(pos_b.x) * 0.05)        _pos_objetivo = Vector3(cam_x, cam_y, cam_z)    global_position = global_position.lerp(_pos_objetivo, delta * 3.8)        var foco_deseado = Vector3(pos_b.x * 0.85, 0.4, pos_b.z * 0.65)    _foco_actual = _foco_actual.lerp(foco_deseado, delta * 5.0)    look_at(_foco_actual, Vector3.UP)
2.2. Locomoción sin Deslizamiento de Pies
Archivo: res://visor/jugador_visor.gdLógica: El Árbol de animación no se activa con estados binarios (correr/parar). Se calcula la velocidad lineal instantánea derivada de la inercia.
# res://visor/jugador_visor.gdclass_name JugadorVisorextends CharacterBody3D@export var velocidad_base_anim_carrera: float = 4.6 @onready var anim_tree: AnimationTree = $AnimationTreevar _posicion_anterior: Vector3 = Vector3.ZEROvar _velocidad_escalar_m_s: float = 0.0func _ready() -> void:    _posicion_anterior = global_positionfunc actualizar_locomocion_tick(delta: float, destino_tactico: Vector3, factor_inercia: float) -> void:    var pos_previa = global_position    global_position = global_position.lerp(destino_tactico, factor_inercia * delta)        var desplazamiento = (global_position - pos_previa)    desplazamiento.y = 0.0    _velocidad_escalar_m_s = desplazamiento.length() / maxf(delta, 0.0001)        if _velocidad_escalar_m_s > 0.35:        var dir = desplazamiento.normalized()        var angulo_rad = atan2(-dir.x, -dir.z)        rotation.y = lerp_angle(rotation.y, angulo_rad, 12.0 * delta)        _sincronizar_animacion()func _sincronizar_animacion() -> void:    var blend: float = clampf(_velocidad_escalar_m_s / 6.5, 0.0, 1.0)    anim_tree.set("parameters/Locomocion/blend_position", blend)        if _velocidad_escalar_m_s > 0.2:        var escala_tiempo = _velocidad_escalar_m_s / velocidad_base_anim_carrera        anim_tree.set("parameters/TimeScale/scale", clampf(escala_tiempo, 0.6, 1.4))    else:        anim_tree.set("parameters/TimeScale/scale", 1.0)

2.4. Banquillos y Entorno Procedural Proporcional
Archivo: res://generadores/stadium_builder.gdLógica: Cálculo estricto del AABB para forzar escala reglamentaria (2.2m de alto) y ubicación 3.5m tras la línea de banda.
# res://generadores/stadium_builder.gdclass_name StadiumBuilderextends Nodeconst ANCHO_CANCHA: float = 68.0const DISTANCIA_LINEA_BANDA: float = 3.5static func configurar_banquillo(instancia: Node3D, es_local: bool) -> void:    var aabb = _obtener_aabb_recursivo(instancia)    if aabb.size.y > 0.001:        var factor_escala = 2.2 / aabb.size.y        instancia.scale = Vector3.ONE * factor_escala        var offset_x = -14.0 if es_local else 14.0    var pos_z = (ANCHO_CANCHA * 0.5) + DISTANCIA_LINEA_BANDA        instancia.global_position = Vector3(offset_x, 0.0, pos_z)    instancia.rotation_degrees.y = 180.0static func _obtener_aabb_recursivo(nodo: Node3D) -> AABB:    var box = AABB()    var primero = true    for hijo in nodo.find_children("*", "MeshInstance3D"):        var mesh_inst = hijo as MeshInstance3D        if mesh_inst and mesh_inst.mesh:            var sub_aabb = mesh_inst.mesh.get_aabb()            if primero:                box = sub_aabb                primero = false            else:                box = box.merge(sub_aabb)    return box
3. Motor Físico Balístico del Balón
Archivo: res://nucleo/balon_fisica.gdModela trayectorias curvas (efecto Magnus), amortiguación de bote sobre césped y sustentación aérea.
# res://nucleo/balon_fisica.gdclass_name BalonFisicaextends Node3Dconst GRAVEDAD: float = 9.81const RESISTENCIA_AIRE: float = 0.12const RESTITUCION_CESPED: float = 0.58var velocidad_vectorial: Vector3 = Vector3.ZEROvar giro_angular: Vector3 = Vector3.ZEROfunc disparar(impulso: Vector3, spin: Vector3) -> void:    velocidad_vectorial = impulso    giro_angular = spinfunc actualizar_fisica(delta: float) -> void:    var fuerza_magnus = giro_angular.cross(velocidad_vectorial) * 0.02    var arrastre = -velocidad_vectorial * RESISTENCIA_AIRE    var aceleracion = Vector3(0, -GRAVEDAD, 0) + fuerza_magnus + arrastre        velocidad_vectorial += aceleracion * delta    global_position += velocidad_vectorial * delta        if global_position.y <= 0.11:         global_position.y = 0.11        velocidad_vectorial.y = -velocidad_vectorial.y * RESTITUCION_CESPED        velocidad_vectorial.x *= 0.82        velocidad_vectorial.z *= 0.82        giro_angular *= 0.7
4. Catálogo Maestro de Jugadas Tácticas Prehechas
Cada jugada se define como un arreglo ordenado de fases cinemáticas que el núcleo evalúa matemáticamente y despacha al MatchPlayback para su renderizado. Se incluyen catalogadas 15 de ataque, 15 de defensa, 15 de portero y 15 de faltas/fueras de juego detalladas en el documento base.
5. Implementación del Gestor de Jugadas
Archivo: res://visor/reproductor_jugadas.gdEste script consume la definición de las 60 jugadas generadas en el núcleo, maneja las fases de interpolación y orquesta tanto la cinemática del balón como las órdenes de cámara.
# res://visor/reproductor_jugadas.gdclass_name ReproductorJugadasextends Nodesignal jugada_iniciada(codigo_jugada: String)signal jugada_completada(resultado: Dictionary)@export var camara_ref: TeleDinamica@export var balon_ref: BalonFisicavar _en_reproduccion: bool = falsevar _tiempo_fase: float = 0.0var _fase_actual: int = 0var _datos_jugada: Dictionary = {}func ejecutar_jugada_prehecha(codigo_jugada: String, datos_actores: Dictionary) -> void:    _datos_jugada = CatalogoJugadas.obtener_definicion(codigo_jugada)    if _datos_jugada.is_empty():        push_error("Jugada no catalogada: " + codigo_jugada)        return        _fase_actual = 0    _tiempo_fase = 0.0    _en_reproduccion = true    emit_signal("jugada_iniciada", codigo_jugada)func _process(delta: float) -> void:    if not _en_reproduccion: return            _tiempo_fase += delta    var fases: Array = _datos_jugada.get("fases", [])        if _fase_actual >= fases.size():        _finalizar_jugada()        return            var fase_data: Dictionary = fases[_fase_actual]    var duracion_fase: float = fase_data.get("duracion", 1.0)        _procesar_fase(fase_data, delta)        if _tiempo_fase >= duracion_fase:        _tiempo_fase = 0.0        _fase_actual += 1func _procesar_fase(fase: Dictionary, delta: float) -> void:    if fase.has("impulso_balon"):        var impulso = fase["impulso_balon"]        var spin = fase.get("spin_balon", Vector3.ZERO)        balon_ref.disparar(impulso, spin)        fase.erase("impulso_balon")        var es_peligro = fase.get("es_peligro", false)    # camara_ref.actualizar_camara(delta, es_peligro) func _finalizar_jugada() -> void:    _en_reproduccion = false    emit_signal("jugada_completada", {"exito": true, "jugada": _datos_jugada.get("codigo", "")})
